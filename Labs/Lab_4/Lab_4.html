import re
from numpy import isin 

def is_veg(x): 
    x=x.lower()
    recipe_list=np.array(x.split(" "))
    is_vegetarian=isin(recipe_list, meats)
    if is_vegetarian.sum()==0:
        return "Vegetarian"
    else: 
        return "Contains meat"
